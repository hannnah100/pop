When a correct answer is submitted:

The corresponding box reveals the answer
Box changes from [?] to the actual title
Box gets a success color (green border/background)
Example: [?] 2008 becomes Iron Man 2008 with green styling

Box Styling:

Unrevealed: Question mark ? symbol, hint text below (year, category, etc.), neutral gray color
Revealed (correct): Full answer text, hint below, green border and background
Grid layout: Responsive (4-5 boxes per row on desktop, 2-3 on mobile)
Sorted by hint: Boxes appear in chronological order (earliest year first)


Data Structure Update:
Each Three Strikes challenge needs:

id: Challenge identifier
date: Date this challenge is active
title: Display title ("Marvel Cinematic Universe Movies")
prompt: What players see ("Name all MCU movies")
totalCount: Number of answers
answers: Array of answer objects with:

correct: Array of accepted variations
hint: The hint text shown below the box
display: The canonical answer to show when revealed

UI Component Code Example:
jsx// ThreeStrikes.jsx component snippet

const ThreeStrikesGame = () => {
  const [answers, setAnswers] = useState([]); // from API
  const [revealed, setRevealed] = useState(new Set()); // indices of correct answers
  const [strikes, setStrikes] = useState(0);
  const [input, setInput] = useState('');
  
  const handleSubmit = (e) => {
    e.preventDefault();
    const guess = input.trim().toLowerCase();
    
    // Check if already guessed
    if (guessedAnswers.has(guess)) {
      showMessage("Already guessed!");
      return;
    }
    
    // Find matching answer
    const matchIndex = answers.findIndex(answer => 
      answer.correct.some(variant => 
        variant.toLowerCase() === guess
      )
    );
    
    if (matchIndex !== -1 && !revealed.has(matchIndex)) {
      // Correct!
      setRevealed(new Set([...revealed, matchIndex]));
      setInput('');
      // Success animation
    } else {
      // Wrong!
      setStrikes(strikes + 1);
      setInput('');
      // Shake animation
      
      if (strikes + 1 >= 3) {
        // Game over
        handleGameOver();
      }
    }
    
    setGuessedAnswers(new Set([...guessedAnswers, guess]));
  };
  
  return (
    <div className="p-6">
      <h1 className="text-3xl font-display mb-2">{challenge.title}</h1>
      <p className="text-pop-cyan mb-6">{challenge.totalCount} total answers</p>
      
      {/* Grid of boxes */}
      <div className="grid grid-cols-3 md:grid-cols-5 lg:grid-cols-6 gap-4 mb-8">
        {answers.map((answer, index) => (
          <div
            key={index}
            className={`
              aspect-square rounded-lg p-3 flex flex-col items-center justify-center
              ${revealed.has(index) 
                ? 'bg-success/20 border-2 border-success' 
                : 'bg-white/10 border-2 border-white/20'
              }
            `}
          >
            <div className="text-2xl font-display mb-1">
              {revealed.has(index) ? '✓' : '?'}
            </div>
            <div className={`text-xs text-center ${revealed.has(index) ? 'text-success' : 'text-white/60'}`}>
              {answer.hint}
            </div>
            {revealed.has(index) && (
              <div className="text-sm text-center mt-2 font-medium">
                {answer.display}
              </div>
            )}
          </div>
        ))}
      </div>
      
      {/* Strikes display */}
      <div className="mb-6 text-center">
        <span className="text-lg">Strikes: </span>
        {[...Array(3)].map((_, i) => (
          <span key={i} className={i < strikes ? 'text-error' : 'text-white/20'}>
            ⚠️
          </span>
        ))}
      </div>
      
      {/* Input */}
      <form onSubmit={handleSubmit} className="max-w-md mx-auto">
        <input
          type="text"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          placeholder="Type your answer..."
          className="input w-full mb-4"
          autoFocus
          disabled={strikes >= 3}
        />
        <button type="submit" className="btn btn-primary w-full">
          Submit
        </button>
      </form>
      
      {/* Progress */}
      <div className="text-center mt-6 text-white/60">
        {revealed.size}/{answers.length} correct
      </div>
    </div>
  );
};
