Writing Phase:

Each player gets assigned someone's "card" (round-robin)
Phone shows: "You're roasting: SARAH"
4 colored text boxes (one per question)
Player picks ONE color to write in (leaves others blank)
Timer: 60-90 seconds
Submit → card passes to next player
Repeat until each player has written on everyone else's card
Each card ends up with one answer per color (from different people)


Reveal Phase:

Each player sees their own card with all roasts
For each color, player:

Reads all answers for that color
Picks favorite answer
Guesses who wrote it (dropdown of other players)


Points awarded:

+1 if your roast was picked as favorite
+1 if you guess correctly who wrote your favorite




Final leaderboard: Best Roaster, Best Detective, Overall Winner
15-20 pre-loaded question sets (pop culture themed, spicy but not mean)

Circular progress ring (like Apple Fitness)

Big numbers in center

Podium style (1st place elevated)
Profile pictures (or initials in colored circles)

Skeleton screens (no spinners)

javascriptconst express = require('express');
const router = express.Router();

// In-memory storage for active rooms
const rooms = new Map();

// Generate unique room code
function generateRoomCode() {
  const words = ['BUZZ', 'ROFL', 'YEET', 'VIBE', 'EPIC', 'FLEX', 'SLAY', 'MOOD'];
  let code;
  do {
    code = words[Math.floor(Math.random() * words.length)];
  } while (rooms.has(code));
  return code;
}

// Create new room
router.post('/create', (req, res) => {
  const { gameType } = req.body; // 'pop-the-question' or 'roast-roulette'
  const roomCode = generateRoomCode();
  
  rooms.set(roomCode, {
    code: roomCode,
    gameType,
    status: 'lobby',
    players: [],
    createdAt: Date.now()
  });
  
  res.json({ roomCode });
});

// Get room info
router.get('/:code', (req, res) => {
  const room = rooms.get(req.params.code);
  if (!room) return res.status(404).json({ error: 'Room not found' });
  res.json(room);
});

socket/events.js:
javascriptconst { rooms } = require('../routes/rooms');

module.exports = (io) => {
  io.on('connection', (socket) => {
    console.log('User connected:', socket.id);
    
    // Join room
    socket.on('join-room', ({ roomCode, playerName, isHost }) => {
      const room = rooms.get(roomCode);
      if (!room) {
        socket.emit('error', { message: 'Room not found' });
        return;
      }
      
      socket.join(roomCode);
      
      const player = {
        id: socket.id,
        name: playerName,
        isHost: isHost || false,
        score: 0
      };
      
      room.players.push(player);
      
      if (isHost) {
        room.hostId = socket.id;
      }
      
      io.to(roomCode).emit('player-joined', {
        player,
        players: room.players
      });
    });
    
    // Start game (host only)
    socket.on('start-game', ({ roomCode }) => {
      const room = rooms.get(roomCode);
      if (!room || room.hostId !== socket.id) return;
      
      room.status = 'playing';
      
      if (room.gameType === 'pop-the-question') {
        // Load first question
        room.currentQuestionIndex = 0;
        room.currentVotes = {};
        // TODO: Load questions from database
        room.questions = [
          "Most likely to become a reality TV star",
          "Most likely to cry during a Pixar movie",
          // ... more questions
        ];
      } else if (room.gameType === 'roast-roulette') {
        // Setup Roast Roulette
        const numQuestions = room.players.length - 1;
        const colors = ['yellow', 'blue', 'red', 'green', 'purple', 'orange', 'gray'];
        // TODO: Load questions from database
        room.questions = []; // Array of {color, text}
        room.currentRound = 1;
        room.cards = {}; // Will store roasts
      }
      
      io.to(roomCode).emit('game-started', {
        gameType: room.gameType,
        status: room.status
      });
    });
    
    // Pop the Question: Submit vote
    socket.on('submit-vote', ({ roomCode, votedForId }) => {
      const room = rooms.get(roomCode);
      if (!room) return;
      
      room.currentVotes[socket.id] = votedForId;
      
      // Check if all votes in
      if (Object.keys(room.currentVotes).length === room.players.length) {
        io.to(roomCode).emit('all-votes-in');
      }
    });
    
    // Pop the Question: Reveal results (host only)
    socket.on('reveal-results', ({ roomCode }) => {
      const room = rooms.get(roomCode);
      if (!room || room.hostId !== socket.id) return;
      
      // Count votes
      const voteCounts = {};
      room.players.forEach(p => voteCounts[p.id] = 0);
      
      Object.values(room.currentVotes).forEach(votedId => {
        voteCounts[votedId]++;
      });
      
      // Update burn counts
      room.players.forEach(player => {
        player.burns = (player.burns || 0) + voteCounts[player.id];
      });
      
      io.to(roomCode).emit('results-revealed', {
        voteCounts,
        players: room.players
      });
      
      // Reset for next round
      room.currentVotes = {};
    });
    
    // Roast Roulette: Submit roast
    socket.on('submit-roast', ({ roomCode, targetPlayerId, color, answer }) => {
      const room = rooms.get(roomCode);
      if (!room) return;
      
      if (!room.cards[targetPlayerId]) {
        room.cards[targetPlayerId] = {};
      }
      
      room.cards[targetPlayerId][color] = {
        author: socket.id,
        answer: answer
      };
      
      // Check if round complete
      // (all players have submitted)
      // TODO: Implement round completion logic
    });
    
    // Disconnect
    socket.on('disconnect', () => {
      console.log('User disconnected:', socket.id);
      
      // Remove from all rooms
      rooms.forEach((room, code) => {
        const playerIndex = room.players.findIndex(p => p.id === socket.id);
        if (playerIndex !== -1) {
          room.players.splice(playerIndex, 1);
          io.to(code).emit('player-left', {
            playerId: socket.id,
            players: room.players
          });
          
          // Delete empty rooms
          if (room.players.length === 0) {
            rooms.delete(code);
          }
        }
      });
    });
  });
};
